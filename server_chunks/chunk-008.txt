route('GET','/api/my/courses',async(req,res)=>{const u=auth(req,res);if(!u)return;const rank=u.role==='admin'?999:userRank(u.id);const rows=db.prepare('SELECT c.*,EXISTS(SELECT 1 FROM course_access ca WHERE ca.user_id=? AND ca.course_id=c.id) direct_access FROM courses c WHERE c.published=1 ORDER BY c.sort_order,c.id').all(u.id).filter(c=>u.role==='admin'||hasCourseAccess(u.id,c.id)).map(c=>({...publicCourse(c),direct_access:!!c.direct_access}));json(res,200,{courses:rows,rank})});
route('GET','/api/my/course/:slug',async(req,res,p,urlObj)=>{const u=auth(req,res);if(!u)return;const c=db.prepare('SELECT * FROM courses WHERE slug=? AND published=1').get(safeString(p.slug,120));if(!c)return json(res,404,{error:'Curso no encontrado.'});if(!hasCourseAccess(u.id,c.id)&&u.role!=='admin')return json(res,403,{error:'Este curso se habilita únicamente después de una compra aprobada o con una membresía que lo incluya.'});const lessons=db.prepare('SELECT l.*,COALESCE(lp.completed,0) completed FROM lessons l LEFT JOIN lesson_progress lp ON lp.lesson_id=l.id AND lp.user_id=? WHERE l.course_id=? AND l.published=1 ORDER BY l.sort_order,l.id').all(u.id,c.id);let rank=userRank(u.id),previewRank=null;if(u.role==='admin'){const q=urlObj.searchParams.get('previewRank');previewRank=q===null?999:Math.max(0,Math.min(2,Number(q)));rank=previewRank}const assets=db.prepare('SELECT id,lesson_id,title,original_name,bytes,required_rank,downloadable FROM assets WHERE course_id=? ORDER BY id DESC').all(c.id).map(a=>({...a,can_download:!!a.downloadable&&rank>=a.required_rank}));json(res,200,{course:publicCourse(c),lessons,assets,preview:u.role==='admin',previewRank})});
route('POST','/api/my/progress/:lessonId',async(req,res,p)=>{const u=auth(req,res);if(!u||!stateGuard(req,res))return;const l=db.prepare('SELECT course_id FROM lessons WHERE id=?').get(Number(p.lessonId));if(!l||(u.role!=='admin'&&!hasCourseAccess(u.id,l.course_id)))return json(res,403,{error:'Sin acceso.'});const b=await readJson(req),completed=b.completed?1:0;db.prepare('INSERT INTO lesson_progress(user_id,lesson_id,completed,updated_at) VALUES(?,?,?,?) ON CONFLICT(user_id,lesson_id) DO UPDATE SET completed=excluded.completed,updated_at=excluded.updated_at').run(u.id,Number(p.lessonId),completed,nowIso());json(res,200,{ok:true})});
