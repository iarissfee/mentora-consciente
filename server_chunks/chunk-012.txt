route('GET','/api/admin/dashboard',async(req,res)=>{if(!auth(req,res,true))return;const students=db.prepare("SELECT COUNT(*) c FROM users WHERE role='student'").get().c,sales=db.prepare("SELECT COUNT(*) c,COALESCE(SUM(amount),0) total FROM sales WHERE status='COMPLETED'").get(),activeMemberships=db.prepare("SELECT COUNT(*) c FROM memberships WHERE active=1 AND datetime(ends_at)>datetime('now')").get().c,courses=db.prepare('SELECT COUNT(*) c FROM courses WHERE EXISTS(SELECT 1 FROM home_programs hp WHERE hp.course_id=courses.id)').get().c,recentSales=db.prepare("SELECT s.*,u.name,u.email,CASE WHEN s.item_type='course' THEN (SELECT title FROM courses WHERE id=s.item_id) ELSE (SELECT name FROM plans WHERE id=s.item_id) END item_name FROM sales s JOIN users u ON u.id=s.user_id ORDER BY s.id DESC LIMIT 20").all();json(res,200,{stats:{students,sales:sales.c,revenue:money(sales.total),activeMemberships,courses},recentSales})});
route('GET','/api/admin/students',async(req,res)=>{if(!auth(req,res,true))return;const rows=db.prepare("SELECT u.id,u.name,u.email,u.created_at,u.last_login_at,p.name membership_name,m.ends_at membership_ends,(SELECT COUNT(*) FROM course_access ca WHERE ca.user_id=u.id) direct_courses FROM users u LEFT JOIN memberships m ON m.id=(SELECT mm.id FROM memberships mm WHERE mm.user_id=u.id AND mm.active=1 AND datetime(mm.ends_at)>datetime('now') ORDER BY mm.id DESC LIMIT 1) LEFT JOIN plans p ON p.id=m.plan_id WHERE u.role='student' ORDER BY u.id DESC").all();json(res,200,{students:rows})});
route('GET','/api/admin/courses',async(req,res)=>{if(!auth(req,res,true))return;const courses=db.prepare('SELECT c.*,(SELECT hp.id FROM home_programs hp WHERE hp.course_id=c.id LIMIT 1) home_program_id,(SELECT hp.active FROM home_programs hp WHERE hp.course_id=c.id LIMIT 1) home_active FROM courses c WHERE EXISTS(SELECT 1 FROM home_programs hp WHERE hp.course_id=c.id) ORDER BY c.sort_order,c.id').all().map(c=>({...c,lessons:db.prepare('SELECT * FROM lessons WHERE course_id=? ORDER BY sort_order,id').all(c.id),assets:db.prepare('SELECT id,lesson_id,title,original_name,bytes,required_rank,downloadable,created_at FROM assets WHERE course_id=? ORDER BY id DESC').all(c.id)}));json(res,200,{courses})});
