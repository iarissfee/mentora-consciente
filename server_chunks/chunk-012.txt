route('GET','/api/admin/dashboard',async(req,res)=>{if(!auth(req,res,true))return;const students=db.prepare("SELECT COUNT(*) c FROM users WHERE role='student'").get().c,sales=db.prepare("SELECT COUNT(*) c,COALESCE(SUM(amount),0) total FROM sales WHERE status='COMPLETED'").get(),activeMemberships=db.prepare("SELECT COUNT(*) c FROM memberships WHERE active=1 AND datetime(ends_at)>datetime('now')").get().c,courses=db.prepare('SELECT COUNT(*) c FROM courses').get().c,recentSales=db.prepare("SELECT s.*,u.name,u.email,CASE WHEN s.item_type='course' THEN (SELECT title FROM courses WHERE id=s.item_id) ELSE (SELECT name FROM plans WHERE id=s.item_id) END item_name FROM sales s JOIN users u ON u.id=s.user_id ORDER BY s.id DESC LIMIT 20").all();json(res,200,{stats:{students,sales:sales.c,revenue:money(sales.total),activeMemberships,courses},recentSales})});
route('GET','/api/admin/students',async(req,res)=>{if(!auth(req,res,true))return;const rows=db.prepare("SELECT u.id,u.name,u.email,u.created_at,u.last_login_at,p.name membership_name,m.ends_at membership_ends,(SELECT COUNT(*) FROM course_access ca WHERE ca.user_id=u.id) direct_courses FROM users u LEFT JOIN memberships m ON m.id=(SELECT mm.id FROM memberships mm WHERE mm.user_id=u.id AND mm.active=1 AND datetime(mm.ends_at)>datetime('now') ORDER BY mm.id DESC LIMIT 1) LEFT JOIN plans p ON p.id=m.plan_id WHERE u.role='student' ORDER BY u.id DESC").all();json(res,200,{students:rows})});

route('POST','/api/admin/courses',async(req,res)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;const b=await readJson(req),title=safeString(b.title,120),slug=safeString(b.slug,120).toLowerCase().replace(/[^a-z0-9-]+/g,'-').replace(/^-|-$/g,'');if(title.length<3||slug.length<3)return json(res,400,{error:'Título y slug son obligatorios.'});try{const t=nowIso(),r=db.prepare('INSERT INTO courses(slug,title,subtitle,description,cover_url,price,currency,required_rank,allow_direct_purchase,published,sort_order,created_at,updated_at) VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?)').run(slug,title,safeString(b.subtitle,180),safeString(b.description,1800),safeString(b.cover_url,600),money(b.price),safeString(b.currency||'USD',3).toUpperCase(),Number(b.required_rank||0),b.allow_direct_purchase?1:0,b.published?1:0,Number(b.sort_order||0),t,t);json(res,200,{ok:true,id:Number(r.lastInsertRowid)})}catch{json(res,409,{error:'Ese slug ya existe o los datos no son válidos.'})}});
