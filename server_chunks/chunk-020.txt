function cleanHomeAssetUrl(v){const s=safeString(v,1200);if(!s)return '';if(s.startsWith('/')&&!s.startsWith('//'))return s;try{const u=new URL(s);if(u.protocol!=='https:')return '';u.username='';u.password='';return u.toString()}catch{return ''}}
function cleanHomeTarget(v){const s=safeString(v,900);if(!s)return '';if(s.startsWith('/')&&!s.startsWith('//'))return s;try{const u=new URL(s);if(u.protocol!=='https:')return '';u.username='';u.password='';return u.toString()}catch{return ''}}
function programSlug(title){return safeString(title,120).normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase().replace(/[^a-z0-9]+/g,'-').replace(/^-|-$/g,'').slice(0,105)||'programa'}
function uniqueCourseSlug(title,ignoreId=0){const base=programSlug(title);let slug=base,n=2;while(db.prepare('SELECT id FROM courses WHERE slug=? AND id<>?').get(slug,ignoreId)){slug=`${base}-${n++}`}return slug}
