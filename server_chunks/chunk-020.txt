
route('PUT','/api/admin/lessons/:id',async(req,res,p)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;const id=Number(p.id),lesson=db.prepare('SELECT * FROM lessons WHERE id=?').get(id);if(!lesson)return json(res,404,{error:'Clase no encontrada.'});const b=await readJson(req),title=safeString(b.title??lesson.title,140),rawVideo=b.video_url===undefined?lesson.video_url:safeString(b.video_url,800),video=rawVideo?cleanLessonVideoUrl(rawVideo):'';if(title.length<2)return json(res,400,{error:'Ingresá un título válido.'});if(rawVideo&&!video)return json(res,400,{error:'El enlace de video no es válido o no pertenece a un proveedor permitido.'});db.prepare('UPDATE lessons SET title=?,body=?,video_url=?,sort_order=?,published=? WHERE id=?').run(title,safeString(b.body??lesson.body,6000),video,Number(b.sort_order??lesson.sort_order),b.published===undefined?lesson.published:(b.published?1:0),id);json(res,200,{ok:true})});
function cleanHomeAssetUrl(v){const s=safeString(v,1200);if(!s)return '';if(s.startsWith('/')&&!s.startsWith('//'))return s;try{const u=new URL(s);if(u.protocol!=='https:')return '';u.username='';u.password='';return u.toString()}catch{return ''}}
function cleanHomeTarget(v){const s=safeString(v,900);if(!s)return '/campus.html#courses';if(s.startsWith('/')&&!s.startsWith('//'))return s;try{const u=new URL(s);if(u.protocol!=='https:')return '';u.username='';u.password='';return u.toString()}catch{return ''}}
function homeSettings(){const s=settingsObj();return{hero_title:s.home_hero_title||'Desprogramate',hero_text:s.home_hero_text||'',video_url:s.home_video_url||'',video_poster:s.home_video_poster||'',programs_title:s.home_programs_title||'Programas',programs_text:s.home_programs_text||''}}
function homeProgramPublic(p){return{id:p.id,title:p.title,description:p.description,price:p.price,currency:p.currency,button_label:p.button_label,target_url:p.target_url,badge:p.badge,style:p.style,sort_order:p.sort_order}}
route('GET','/api/public/home',async(req,res)=>{const programs=db.prepare('SELECT * FROM home_programs WHERE active=1 ORDER BY sort_order,id').all().map(homeProgramPublic);json(res,200,{settings:homeSettings(),programs})});

route('PUT','/api/admin/home',async(req,res)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;const b=await readJson(req),videoRaw=safeString(b.video_url,900),video=videoRaw?cleanLessonVideoUrl(videoRaw):'',poster=cleanHomeAssetUrl(b.video_poster);if(videoRaw&&!video)return json(res,400,{error:'El video debe ser un enlace HTTPS válido de YouTube, Vimeo, Flowplayer/Wowza o un MP4 compatible.'});if(b.video_poster&&!poster)return json(res,400,{error:'La portada del video debe ser una URL HTTPS válida.'});const values={home_hero_title:safeString(b.hero_title,120),home_hero_text:safeString(b.hero_text,500),home_video_url:video,home_video_poster:poster,home_programs_title:safeString(b.programs_title,100),home_programs_text:safeString(b.programs_text,500)},q=db.prepare('INSERT INTO settings(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value');tx(()=>{for(const [k,v] of Object.entries(values))q.run(k,v)});json(res,200,{ok:true})});
route('POST','/api/admin/home/programs',async(req,res)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;const b=await readJson(req),title=safeString(b.title,120),target=cleanHomeTarget(b.target_url),style=['standard','popular','club'].includes(b.style)?b.style:'standard';if(title.length<2)return json(res,400,{error:'Ingresá un título para la tarjeta.'});if(!target)return json(res,400,{error:'El destino de la tarjeta no es válido.'});const t=nowIso(),r=db.prepare('INSERT INTO home_programs(title,description,price,currency,button_label,target_url,badge,style,active,sort_order,created_at,updated_at) VALUES(?,?,?,?,?,?,?,?,?,?,?,?)').run(title,safeString(b.description,700),Math.max(0,Number(b.price||0)),safeString(b.currency||'EUR',3).toUpperCase(),safeString(b.button_label||'Ver programa',80),target,safeString(b.badge,30),style,b.active===false?0:1,Number(b.sort_order||0),t,t);json(res,200,{ok:true,id:Number(r.lastInsertRowid)})});
