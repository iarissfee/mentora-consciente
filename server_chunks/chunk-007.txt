function stateGuard(req,res){if(!checkCsrf(req)){json(res,403,{error:'Solicitud rechazada. Actualizá la página e intentá nuevamente.'});return false}return true}

route('GET','/api/csrf',async(req,res)=>json(res,200,{token:csrfToken(req,res)}));
route('GET','/api/health',async(req,res)=>json(res,200,{ok:true}));
route('GET','/api/config',async(req,res)=>{csrfToken(req,res);const id=process.env.PAYPAL_CLIENT_ID||'';json(res,200,{settings:settingsObj(),paypal:{enabled:!!(id&&process.env.PAYPAL_CLIENT_SECRET),clientId:id,mode:process.env.PAYPAL_MODE==='live'?'live':'sandbox'},demoCheckout:!(id&&process.env.PAYPAL_CLIENT_SECRET)&&!isProd})});
route('GET','/api/catalog',async(req,res)=>json(res,200,{plans:db.prepare('SELECT * FROM plans WHERE active=1 ORDER BY rank').all().map(publicPlan),courses:db.prepare('SELECT * FROM courses WHERE published=1 ORDER BY sort_order,id').all().map(publicCourse)}));
route('GET','/api/me',async(req,res)=>{const u=sessionUser(req);if(!u)return json(res,200,{user:null});json(res,200,{user:{id:u.id,name:u.name,email:u.email,role:u.role},membership:activeMembership(u.id),directCourseIds:db.prepare('SELECT course_id FROM course_access WHERE user_id=?').all(u.id).map(x=>x.course_id)})});

route('POST','/api/auth/register',async(req,res)=>{if(!rateOk(`reg:${clientIp(req)}`,20,15*60000))return json(res,429,{error:'Demasiados intentos. Probá más tarde.'});if(!stateGuard(req,res))return;const b=await readJson(req),name=safeString(b.name,80),email=safeEmail(b.email),password=String(b.password||'');if(name.length<2)return json(res,400,{error:'Ingresá tu nombre.'});if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email))return json(res,400,{error:'Ingresá un email válido.'});if(password.length<10||password.length>128)return json(res,400,{error:'La contraseña debe tener entre 10 y 128 caracteres.'});if(db.prepare('SELECT id FROM users WHERE email=?').get(email))return json(res,409,{error:'Ese email ya está registrado.'});const r=db.prepare('INSERT INTO users(name,email,password_hash,role,created_at) VALUES(?,?,?,?,?)').run(name,email,hashPassword(password),'student',nowIso());createSession(Number(r.lastInsertRowid),res);json(res,200,{ok:true})});
route('POST','/api/auth/login',async(req,res)=>{if(!rateOk(`login:${clientIp(req)}`,20,15*60000))return json(res,429,{error:'Demasiados intentos. Probá más tarde.'});if(!stateGuard(req,res))return;const b=await readJson(req),email=safeEmail(b.email),password=String(b.password||''),u=db.prepare('SELECT * FROM users WHERE email=?').get(email);if(!u||!verifyPassword(password,u.password_hash)){await sleep(250);return json(res,401,{error:'Email o contraseña incorrectos.'})}db.prepare('UPDATE users SET last_login_at=? WHERE id=?').run(nowIso(),u.id);createSession(u.id,res);json(res,200,{ok:true,role:u.role})});
route('POST','/api/auth/logout',async(req,res)=>{if(!stateGuard(req,res))return;const raw=parseCookies(req).mc_session;if(raw)db.prepare('DELETE FROM sessions WHERE token_hash=?').run(sha256(raw));clearCookie(res,'mc_session');json(res,200,{ok:true})});
