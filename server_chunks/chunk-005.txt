function json(res,status,obj){securityHeaders(res);const b=Buffer.from(JSON.stringify(obj));res.writeHead(status,{'Content-Type':'application/json; charset=utf-8','Content-Length':b.length,'Cache-Control':'no-store'});res.end(b)}
function text(res,status,s,type='text/plain; charset=utf-8'){securityHeaders(res);const b=Buffer.from(s);res.writeHead(status,{'Content-Type':type,'Content-Length':b.length});res.end(b)}
async function readBody(req,max=250000){let total=0,chunks=[];for await(const c of req){total+=c.length;if(total>max)throw Object.assign(new Error('Body too large'),{status:413});chunks.push(c)}return Buffer.concat(chunks)}
async function readJson(req,max=250000){const b=await readBody(req,max);if(!b.length)return{};try{return JSON.parse(b.toString('utf8'))}catch{throw Object.assign(new Error('JSON inválido'),{status:400})}}
function csrfToken(req,res){const c=parseCookies(req);let t=c.mc_csrf;if(!/^[a-f0-9]{64}$/.test(t||'')){t=randomToken(32);setCookie(res,'mc_csrf',t,{httpOnly:false,sameSite:'Strict',secure:isProd,maxAge:7*86400000})}return t}
function checkCsrf(req){const c=parseCookies(req).mc_csrf||'',h=String(req.headers['x-csrf-token']||'');return c.length===64&&h.length===64&&crypto.timingSafeEqual(Buffer.from(c),Buffer.from(h))}
function sessionUser(req){const raw=parseCookies(req).mc_session;if(!raw)return null;const row=db.prepare('SELECT u.id,u.name,u.email,u.role,s.expires_at FROM sessions s JOIN users u ON u.id=s.user_id WHERE s.token_hash=?').get(sha256(raw));if(!row)return null;if(new Date(row.expires_at)<=new Date()){db.prepare('DELETE FROM sessions WHERE token_hash=?').run(sha256(raw));return null}return row}
function createSession(userId,res){const raw=randomToken(32),days=Math.max(1,Math.min(90,Number(process.env.SESSION_DAYS||14)));db.prepare('INSERT INTO sessions(user_id,token_hash,expires_at,created_at) VALUES(?,?,?,?)').run(userId,sha256(raw),addDaysIso(days),nowIso());setCookie(res,'mc_session',raw,{httpOnly:true,sameSite:'Lax',secure:isProd,maxAge:days*86400000})}
function activeMembership(userId){return db.prepare("SELECT m.*,p.name plan_name,p.rank,p.slug,p.description FROM memberships m JOIN plans p ON p.id=m.plan_id WHERE m.user_id=? AND m.active=1 AND datetime(m.ends_at)>datetime('now') ORDER BY p.rank DESC,datetime(m.ends_at) DESC LIMIT 1").get(userId)||null}
function userRank(id){return activeMembership(id)?.rank||0}
function hasCourseAccess(userId,courseId){const c=db.prepare('SELECT id,required_rank FROM courses WHERE id=?').get(courseId);if(!c)return false;return !!db.prepare('SELECT id FROM course_access WHERE user_id=? AND course_id=?').get(userId,courseId)||userRank(userId)>=c.required_rank}
