route('POST','/api/demo/purchase',async(req,res)=>{const u=auth(req,res);if(!u||!stateGuard(req,res))return;if(isProd||(process.env.PAYPAL_CLIENT_ID&&process.env.PAYPAL_CLIENT_SECRET))return json(res,404,{error:'No disponible.'});const b=await readJson(req),type=safeString(b.itemType,20),id=Number(b.itemId),item=itemForPurchase(type,id);if(!item)return json(res,400,{error:'Producto no disponible.'});const ref=`DEMO-${Date.now()}-${crypto.randomBytes(3).toString('hex')}`;grantPurchase(u.id,type,id,'demo',ref,item.price,item.currency);let redirect='/#campus';if(type==='course'){const c=db.prepare('SELECT slug FROM courses WHERE id=?').get(id);if(c)redirect=`/#course/${c.slug}`}json(res,200,{ok:true,redirect})});
route('POST','/api/paypal/create-order',async(req,res)=>{const u=auth(req,res);if(!u||!stateGuard(req,res))return;if(!rateOk(`pay:${u.id}`,20,60000))return json(res,429,{error:'Demasiados intentos de pago.'});try{const b=await readJson(req),type=safeString(b.itemType,20),id=Number(b.itemId),item=itemForPurchase(type,id);if(!item||item.price<=0)return json(res,400,{error:'Producto no disponible.'});const {token,base}=await paypalToken();const rr=await fetch(`${base}/v2/checkout/orders`,{method:'POST',headers:{Authorization:`Bearer ${token}`,'Content-Type':'application/json','PayPal-Request-Id':crypto.randomUUID()},body:JSON.stringify({intent:'CAPTURE',purchase_units:[{reference_id:`${type}-${id}`,description:item.name.slice(0,120),amount:{currency_code:item.currency,value:item.price.toFixed(2)}}]})});const data=await rr.json();if(!rr.ok||!data.id)return json(res,502,{error:'No se pudo iniciar el pago con PayPal.'});db.prepare('INSERT INTO pending_orders(user_id,item_type,item_id,amount,currency,provider_order_id,status,created_at) VALUES(?,?,?,?,?,?,?,?)').run(u.id,type,id,item.price,item.currency,data.id,'CREATED',nowIso());json(res,200,{id:data.id})}catch(e){console.error(e);json(res,503,{error:'PayPal todavía no está configurado o no respondió.'})}});
