
// Portada única y editable de Home. Soporta JPG/PNG/WebP y usa un JPG local real como respaldo.
const __homeSettingsOriginalPoster=homeSettings;
homeSettings=function(){
  const out=__homeSettingsOriginalPoster();
  const s=settingsObj(),stored=String(s.home_video_poster_stored_name||'');
  out.video_poster_uploaded=!!stored;
  if(stored)out.video_poster=`/api/home/uploaded-poster?v=${encodeURIComponent(s.home_video_poster_updated_at||stored)}`;
  else out.video_poster='/mentora-original-hero.jpg';
  return out
};

function mcHomePosterMime(name){
  const ext=path.extname(name||'').toLowerCase();
  if(ext==='.png')return'image/png';
  if(ext==='.webp')return'image/webp';
  return'image/jpeg'
}
function mcValidHomePoster(f){
  if(!f||!f.data||!f.data.length)return null;
  const type=String(f.contentType||'').toLowerCase(),d=f.data;
  if(type==='image/jpeg'&&d.length>3&&d[0]===0xff&&d[1]===0xd8&&d[2]===0xff)return{ext:'.jpg',mime:'image/jpeg'};
  if(type==='image/png'&&d.length>8&&d[0]===0x89&&d.slice(1,4).toString()==='PNG')return{ext:'.png',mime:'image/png'};
  if(type==='image/webp'&&d.length>12&&d.slice(0,4).toString()==='RIFF'&&d.slice(8,12).toString()==='WEBP')return{ext:'.webp',mime:'image/webp'};
  return null
}
function mcServeFallbackPoster(res){
  const fp=path.join(PUBLIC_DIR,'mentora-original-hero.jpg');
  if(!fs.existsSync(fp))return json(res,404,{error:'Portada de respaldo no disponible.'});
  const st=fs.statSync(fp);securityHeaders(res);
  res.writeHead(200,{'Content-Type':'image/jpeg','Content-Length':st.size,'Cache-Control':'public, max-age=86400','Content-Disposition':'inline'});
  fs.createReadStream(fp).pipe(res)
}

route('GET','/api/home/original-poster',async(req,res)=>mcServeFallbackPoster(res));

route('GET','/api/home/uploaded-poster',async(req,res)=>{
  const s=settingsObj(),stored=path.basename(String(s.home_video_poster_stored_name||''));
  if(!stored)return mcServeFallbackPoster(res);
  const fp=path.join(UPLOAD_DIR,stored);
  if(!fs.existsSync(fp)&&remotePool){
    try{
      const rr=await remotePool.query('SELECT data FROM mc_asset_blobs WHERE stored_name=$1',[stored]);
      const data=rr.rows[0]?.data;if(data)fs.writeFileSync(fp,data,{mode:0o600});
    }catch(e){console.error('Recuperación portada Home:',e)}
  }
  if(!fs.existsSync(fp))return mcServeFallbackPoster(res);
  const st=fs.statSync(fp);securityHeaders(res);
  res.writeHead(200,{'Content-Type':mcHomePosterMime(stored),'Content-Length':st.size,'Cache-Control':'no-cache','Content-Disposition':'inline'});
  fs.createReadStream(fp).pipe(res)
});

route('POST','/api/admin/home/poster',async(req,res)=>{
  if(!auth(req,res,true)||!stateGuard(req,res))return;
  const ct=String(req.headers['content-type']||'');
  if(!ct.startsWith('multipart/form-data'))return json(res,400,{error:'Formato de carga inválido.'});
  let buf;try{buf=await readBody(req,20*1024*1024+350000)}catch{return json(res,400,{error:'La imagen supera el límite de 20 MB.'})}
  const {files}=parseMultipart(buf,ct),f=files.poster;
  if(!f)return json(res,400,{error:'Elegí una foto para la portada.'});
  if(f.data.length>20*1024*1024)return json(res,400,{error:'La imagen supera el límite de 20 MB.'});
  const info=mcValidHomePoster(f);
  if(!info)return json(res,400,{error:'Usá una imagen JPG, PNG o WebP válida.'});
  const s=settingsObj(),old=path.basename(String(s.home_video_poster_stored_name||''));
  const stored=`home-poster-${Date.now()}-${crypto.randomBytes(8).toString('hex')}${info.ext}`;
  fs.writeFileSync(path.join(UPLOAD_DIR,stored),f.data,{mode:0o600});
  if(old&&old!==stored){const oldPath=path.join(UPLOAD_DIR,old);if(fs.existsSync(oldPath))fs.unlinkSync(oldPath)}
  const t=nowIso(),url=`/api/home/uploaded-poster?v=${encodeURIComponent(t)}`,q=db.prepare('INSERT INTO settings(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value');
  tx(()=>{q.run('home_video_poster_stored_name',stored);q.run('home_video_poster_updated_at',t);q.run('home_video_poster',url)});
  json(res,200,{ok:true,video_poster:url})
});

route('DELETE','/api/admin/home/poster',async(req,res)=>{
  if(!auth(req,res,true)||!stateGuard(req,res))return;
  const s=settingsObj(),old=path.basename(String(s.home_video_poster_stored_name||''));
  if(old){const fp=path.join(UPLOAD_DIR,old);if(fs.existsSync(fp))fs.unlinkSync(fp)}
  const q=db.prepare('INSERT INTO settings(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value'),t=nowIso();
  tx(()=>{q.run('home_video_poster_stored_name','');q.run('home_video_poster_updated_at',t);q.run('home_video_poster','/mentora-original-hero.jpg')});
  json(res,200,{ok:true,video_poster:'/mentora-original-hero.jpg'})
});
