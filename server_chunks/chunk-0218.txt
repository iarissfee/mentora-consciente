
// Portada única y editable de Home.
// La imagen subida se guarda dentro de settings (SQLite persistido en PostgreSQL),
// evitando depender del filesystem efímero de Render.
const __homeSettingsOriginalPoster=homeSettings;
homeSettings=function(){
  const out=__homeSettingsOriginalPoster();
  const s=settingsObj();
  const data=String(s.home_video_poster_data||'');
  const stored=String(s.home_video_poster_stored_name||'');
  out.video_poster_uploaded=!!(data||stored);
  if(data){
    out.video_poster=`/api/home/uploaded-poster?v=${encodeURIComponent(s.home_video_poster_updated_at||'1')}`;
  }else if(stored){
    out.video_poster=`/api/home/uploaded-poster?v=${encodeURIComponent(s.home_video_poster_updated_at||stored)}`;
  }else{
    out.video_poster='/api/home/original-poster?v=4';
  }
  return out
};

function mcHomePosterMime(name){
  const ext=path.extname(name||'').toLowerCase();
  if(ext==='.png')return'image/png';
  if(ext==='.webp')return'image/webp';
  return'image/jpeg'
}
function mcValidHomePoster(f){
  if(!f||!f.data||!f.data.length)return null;
  const type=String(f.contentType||'').toLowerCase(),d=f.data;
  if(type==='image/jpeg'&&d.length>3&&d[0]===0xff&&d[1]===0xd8&&d[2]===0xff)return{ext:'.jpg',mime:'image/jpeg'};
  if(type==='image/png'&&d.length>8&&d[0]===0x89&&d.slice(1,4).toString()==='PNG')return{ext:'.png',mime:'image/png'};
  if(type==='image/webp'&&d.length>12&&d.slice(0,4).toString()==='RIFF'&&d.slice(8,12).toString()==='WEBP')return{ext:'.webp',mime:'image/webp'};
  return null
}
function mcServeFallbackPoster(res){
  // Primero intenta un JPG real si alguna vez se agrega al repo.
  const jpg=path.join(PUBLIC_DIR,'mentora-original-hero.jpg');
  if(fs.existsSync(jpg)){
    const st=fs.statSync(jpg);securityHeaders(res);
    res.writeHead(200,{'Content-Type':'image/jpeg','Content-Length':st.size,'Cache-Control':'public, max-age=86400','Content-Disposition':'inline'});
    return fs.createReadStream(jpg).pipe(res)
  }
  // El respaldo que sí existe hoy está guardado como base64.
  const b64File=path.join(PUBLIC_DIR,'mentora-original-hero.jpg.b64');
  if(fs.existsSync(b64File)){
    try{
      const raw=fs.readFileSync(b64File,'utf8').trim(),buf=Buffer.from(raw,'base64');
      if(buf.length>3&&buf[0]===0xff&&buf[1]===0xd8&&buf[2]===0xff){
        securityHeaders(res);
        res.writeHead(200,{'Content-Type':'image/jpeg','Content-Length':buf.length,'Cache-Control':'public, max-age=86400','Content-Disposition':'inline'});
        return res.end(buf)
      }
    }catch(e){console.error('Portada original Mentora:',e)}
  }
  return json(res,404,{error:'Portada de respaldo no disponible.'})
}

route('GET','/api/home/original-poster',async(req,res)=>mcServeFallbackPoster(res));

route('GET','/api/home/uploaded-poster',async(req,res)=>{
  const s=settingsObj();
  const data=String(s.home_video_poster_data||'');
  if(data){
    try{
      const buf=Buffer.from(data,'base64'),mime=String(s.home_video_poster_mime||'image/jpeg');
      if(!buf.length)throw new Error('Imagen vacía');
      securityHeaders(res);
      res.writeHead(200,{
        'Content-Type':['image/jpeg','image/png','image/webp'].includes(mime)?mime:'image/jpeg',
        'Content-Length':buf.length,
        'Cache-Control':'no-cache, no-store, must-revalidate',
        'Content-Disposition':'inline'
      });
      return res.end(buf)
    }catch(e){console.error('Portada Home DB:',e)}
  }

  // Compatibilidad con cargas anteriores guardadas como archivo.
  const stored=path.basename(String(s.home_video_poster_stored_name||''));
  if(!stored)return mcServeFallbackPoster(res);
  const fp=path.join(UPLOAD_DIR,stored);
  if(!fs.existsSync(fp)&&remotePool){
    try{
      const rr=await remotePool.query('SELECT data FROM mc_asset_blobs WHERE stored_name=$1',[stored]);
      const oldData=rr.rows[0]?.data;if(oldData)fs.writeFileSync(fp,oldData,{mode:0o600});
    }catch(e){console.error('Recuperación portada Home:',e)}
  }
  if(!fs.existsSync(fp))return mcServeFallbackPoster(res);
  const st=fs.statSync(fp);securityHeaders(res);
  res.writeHead(200,{'Content-Type':mcHomePosterMime(stored),'Content-Length':st.size,'Cache-Control':'no-cache, no-store, must-revalidate','Content-Disposition':'inline'});
  fs.createReadStream(fp).pipe(res)
});

route('POST','/api/admin/home/poster',async(req,res)=>{
  if(!auth(req,res,true)||!stateGuard(req,res))return;
  const ct=String(req.headers['content-type']||'');
  if(!ct.startsWith('multipart/form-data'))return json(res,400,{error:'Formato de carga inválido.'});
  let buf;try{buf=await readBody(req,8*1024*1024+350000)}catch{return json(res,400,{error:'La imagen supera el límite de 8 MB.'})}
  const {files}=parseMultipart(buf,ct),f=files.poster;
  if(!f)return json(res,400,{error:'Elegí una foto para la portada.'});
  if(f.data.length>8*1024*1024)return json(res,400,{error:'La imagen supera el límite de 8 MB.'});
  const info=mcValidHomePoster(f);
  if(!info)return json(res,400,{error:'Usá una imagen JPG, PNG o WebP válida.'});

  const t=nowIso(),q=db.prepare('INSERT INTO settings(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value');
  tx(()=>{
    q.run('home_video_poster_data',f.data.toString('base64'));
    q.run('home_video_poster_mime',info.mime);
    q.run('home_video_poster_updated_at',t);
    q.run('home_video_poster_stored_name','');
    q.run('home_video_poster',`/api/home/uploaded-poster?v=${encodeURIComponent(t)}`);
  });
  json(res,200,{ok:true,video_poster:`/api/home/uploaded-poster?v=${encodeURIComponent(t)}`})
});

route('DELETE','/api/admin/home/poster',async(req,res)=>{
  if(!auth(req,res,true)||!stateGuard(req,res))return;
  const s=settingsObj(),old=path.basename(String(s.home_video_poster_stored_name||''));
  if(old){const fp=path.join(UPLOAD_DIR,old);if(fs.existsSync(fp))fs.unlinkSync(fp)}
  const q=db.prepare('INSERT INTO settings(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value'),t=nowIso();
  tx(()=>{
    q.run('home_video_poster_data','');
    q.run('home_video_poster_mime','');
    q.run('home_video_poster_stored_name','');
    q.run('home_video_poster_updated_at',t);
    q.run('home_video_poster','/api/home/original-poster?v=4');
  });
  json(res,200,{ok:true,video_poster:'/api/home/original-poster?v=4'})
});
