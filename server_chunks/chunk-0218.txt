
// Portada de Home: permite JPG subido desde Administración y conserva una portada local de respaldo.
const __homeSettingsOriginalPoster=homeSettings;
homeSettings=function(){
  const out=__homeSettingsOriginalPoster();
  const s=settingsObj(),stored=String(s.home_video_poster_stored_name||'');
  out.video_poster_uploaded=!!stored;
  if(stored)out.video_poster=`/api/home/uploaded-poster?v=${encodeURIComponent(s.home_video_poster_updated_at||'1')}`;
  else if(!out.video_poster)out.video_poster='/api/home/original-poster';
  return out
};

route('GET','/api/home/original-poster',async(req,res)=>{
  try{
    const b64=fs.readFileSync(path.join(PUBLIC_DIR,'mentora-original-hero.jpg.b64'),'utf8').trim();
    const buf=Buffer.from(b64,'base64');
    securityHeaders(res);
    res.writeHead(200,{'Content-Type':'image/jpeg','Content-Length':buf.length,'Cache-Control':'public, max-age=86400'});
    res.end(buf)
  }catch(e){
    console.error('Portada original Mentora:',e);
    json(res,500,{error:'No se pudo cargar la portada original.'})
  }
});

route('GET','/api/home/uploaded-poster',async(req,res)=>{
  const s=settingsObj(),stored=path.basename(String(s.home_video_poster_stored_name||''));
  if(!stored)return json(res,404,{error:'No hay una foto subida para la Home.'});
  const fp=path.join(UPLOAD_DIR,stored);
  if(!fs.existsSync(fp)&&remotePool){
    try{
      const rr=await remotePool.query('SELECT data FROM mc_asset_blobs WHERE stored_name=$1',[stored]);
      const data=rr.rows[0]?.data;if(data)fs.writeFileSync(fp,data,{mode:0o600});
    }catch(e){console.error('Recuperación portada Home:',e)}
  }
  if(!fs.existsSync(fp))return json(res,404,{error:'La foto de Home no está disponible.'});
  const st=fs.statSync(fp);securityHeaders(res);
  res.writeHead(200,{'Content-Type':'image/jpeg','Content-Length':st.size,'Cache-Control':'public, max-age=86400','Content-Disposition':'inline'});
  fs.createReadStream(fp).pipe(res)
});

route('POST','/api/admin/home/poster',async(req,res)=>{
  if(!auth(req,res,true)||!stateGuard(req,res))return;
  const ct=String(req.headers['content-type']||'');
  if(!ct.startsWith('multipart/form-data'))return json(res,400,{error:'Formato de carga inválido.'});
  let buf;try{buf=await readBody(req,5*1024*1024+200000)}catch{return json(res,400,{error:'La imagen supera el límite de 5 MB.'})}
  const {files}=parseMultipart(buf,ct),f=files.poster;
  if(!f)return json(res,400,{error:'Elegí primero una imagen JPG.'});
  if(f.data.length>5*1024*1024)return json(res,400,{error:'La imagen supera el límite de 5 MB.'});
  const type=String(f.contentType||'').toLowerCase(),ext=path.extname(f.filename||'').toLowerCase();
  const valid=type==='image/jpeg'&&['.jpg','.jpeg'].includes(ext)&&f.data.length>3&&f.data[0]===0xff&&f.data[1]===0xd8&&f.data[2]===0xff;
  if(!valid)return json(res,400,{error:'Usá una imagen JPG o JPEG válida.'});
  const s=settingsObj(),old=path.basename(String(s.home_video_poster_stored_name||''));
  const stored=`home-poster-${Date.now()}-${crypto.randomBytes(8).toString('hex')}.jpg`;
  fs.writeFileSync(path.join(UPLOAD_DIR,stored),f.data,{mode:0o600});
  if(old&&old!==stored){const oldPath=path.join(UPLOAD_DIR,old);if(fs.existsSync(oldPath))fs.unlinkSync(oldPath)}
  const t=nowIso(),q=db.prepare('INSERT INTO settings(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value');
  tx(()=>{q.run('home_video_poster_stored_name',stored);q.run('home_video_poster_updated_at',t);q.run('home_video_poster','/api/home/uploaded-poster')});
  json(res,200,{ok:true,video_poster:`/api/home/uploaded-poster?v=${encodeURIComponent(t)}`})
});

route('DELETE','/api/admin/home/poster',async(req,res)=>{
  if(!auth(req,res,true)||!stateGuard(req,res))return;
  const s=settingsObj(),old=path.basename(String(s.home_video_poster_stored_name||''));
  if(old){const fp=path.join(UPLOAD_DIR,old);if(fs.existsSync(fp))fs.unlinkSync(fp)}
  const q=db.prepare('INSERT INTO settings(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value');
  tx(()=>{q.run('home_video_poster_stored_name','');q.run('home_video_poster_updated_at',nowIso());q.run('home_video_poster','')});
  json(res,200,{ok:true})
});
