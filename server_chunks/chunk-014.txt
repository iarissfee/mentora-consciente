route('POST','/api/admin/grant',async(req,res)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;const b=await readJson(req),userId=Number(b.userId),type=safeString(b.type,20),itemId=Number(b.itemId);if(!db.prepare("SELECT id FROM users WHERE id=? AND role='student'").get(userId))return json(res,404,{error:'Alumno no encontrado.'});if(type==='course'){if(!db.prepare('SELECT id FROM courses WHERE id=?').get(itemId))return json(res,404,{error:'Curso no encontrado.'});db.prepare("INSERT INTO course_access(user_id,course_id,source,source_ref,granted_at) VALUES(?,?,?,?,?) ON CONFLICT(user_id,course_id) DO NOTHING").run(userId,itemId,'admin','manual',nowIso())}else if(type==='ebook'){if(!db.prepare('SELECT id FROM ebooks WHERE id=?').get(itemId))return json(res,404,{error:'Ebook no encontrado.'});db.prepare("INSERT INTO ebook_access(user_id,ebook_id,source,source_ref,granted_at) VALUES(?,?,?,?,?) ON CONFLICT(user_id,ebook_id) DO UPDATE SET source=excluded.source,source_ref=excluded.source_ref,granted_at=excluded.granted_at").run(userId,itemId,'admin','manual',nowIso())}else if(type==='plan'){const plan=db.prepare('SELECT * FROM plans WHERE id=?').get(itemId);if(!plan)return json(res,404,{error:'Plan no encontrado.'});db.prepare('UPDATE memberships SET active=0 WHERE user_id=?').run(userId);db.prepare('INSERT INTO memberships(user_id,plan_id,starts_at,ends_at,source,source_ref,active) VALUES(?,?,?,?,?,?,1)').run(userId,itemId,nowIso(),addDaysIso(plan.duration_days),'admin','manual')}else return json(res,400,{error:'Tipo inválido.'});json(res,200,{ok:true})});
