
function meditationPublicRow(m){return{id:m.id,title:m.title,description:m.description,sort_order:m.sort_order,qr_url:`/api/community/meditations/${m.id}/qr`}}
route('GET','/api/community/meditations',async(req,res)=>{const u=auth(req,res);if(!u)return;const rows=db.prepare('SELECT * FROM community_meditations WHERE active=1 ORDER BY sort_order,id').all().map(meditationPublicRow);json(res,200,{meditations:rows})});
route('GET','/api/community/meditations/:id/qr',async(req,res,p)=>{const u=auth(req,res);if(!u)return;const m=db.prepare('SELECT * FROM community_meditations WHERE id=?').get(Number(p.id));if(!m||(!m.active&&u.role!=='admin'))return json(res,404,{error:'Meditación no encontrada.'});try{const png=await QRCode.toBuffer(m.youtube_url,{type:'png',width:420,margin:2,errorCorrectionLevel:'M'});securityHeaders(res);res.writeHead(200,{'Content-Type':'image/png','Content-Length':png.length,'Cache-Control':'private, max-age=300','Content-Disposition':'inline'});res.end(png)}catch(e){console.error(e);json(res,500,{error:'No se pudo generar el QR.'})}});
route('GET','/api/admin/meditations',async(req,res)=>{if(!auth(req,res,true))return;const rows=db.prepare('SELECT * FROM community_meditations ORDER BY sort_order,id').all();json(res,200,{meditations:rows})});
route('POST','/api/admin/meditations',async(req,res)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;const b=await readJson(req),title=safeString(b.title,120),url=cleanYoutubeUrl(b.youtube_url),description=safeString(b.description,500);if(title.length<2)return json(res,400,{error:'Ingresá un título.'});if(!url)return json(res,400,{error:'Pegá un enlace válido de YouTube.'});const t=nowIso(),r=db.prepare('INSERT INTO community_meditations(title,youtube_url,description,active,sort_order,created_at,updated_at) VALUES(?,?,?,?,?,?,?)').run(title,url,description,b.active===false?0:1,Number(b.sort_order||0),t,t);json(res,200,{ok:true,id:Number(r.lastInsertRowid)})});
route('PUT','/api/admin/meditations/:id',async(req,res,p)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;const id=Number(p.id),m=db.prepare('SELECT * FROM community_meditations WHERE id=?').get(id);if(!m)return json(res,404,{error:'Meditación no encontrada.'});const b=await readJson(req),title=safeString(b.title??m.title,120),raw=b.youtube_url===undefined?m.youtube_url:b.youtube_url,url=cleanYoutubeUrl(raw);if(title.length<2||!url)return json(res,400,{error:'Título o enlace de YouTube inválido.'});db.prepare('UPDATE community_meditations SET title=?,youtube_url=?,description=?,active=?,sort_order=?,updated_at=? WHERE id=?').run(title,url,safeString(b.description??m.description,500),b.active===undefined?m.active:(b.active?1:0),Number(b.sort_order??m.sort_order),nowIso(),id);json(res,200,{ok:true})});
route('DELETE','/api/admin/meditations/:id',async(req,res,p)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;db.prepare('DELETE FROM community_meditations WHERE id=?').run(Number(p.id));json(res,200,{ok:true})});
