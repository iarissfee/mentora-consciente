
// Portadas estables: usa imágenes incluidas en el repo para los ebooks base.
const __localEbookCovers={};
try{
  __localEbookCovers['el-poder-del-ahora']='data:image/webp;base64,'+fs.readFileSync(path.join(PUBLIC_DIR,'ebook-cover-el-poder.webp.b64'),'utf8').trim();
  __localEbookCovers['guia-de-descompresion']='data:image/webp;base64,'+fs.readFileSync(path.join(PUBLIC_DIR,'ebook-cover-guia-descompresion.webp.b64'),'utf8').trim();
  __localEbookCovers['auditoria-interna']='data:image/webp;base64,'+fs.readFileSync(path.join(PUBLIC_DIR,'ebook-cover-auditoria.webp.b64'),'utf8').trim();
}catch(e){console.warn('No se pudieron cargar portadas locales de ebooks',e.message)}

ebookCoverUrl=function(x){
  if(x?.cover_stored_name)return `/api/ebooks/${x.id}/cover?v=${encodeURIComponent(x.updated_at||'')}`;
  const slug=String(x?.slug||'').toLowerCase();
  if(__localEbookCovers[slug])return __localEbookCovers[slug];
  if(slug==='sana-tu-cuerpo'||slug==='5x5-ludwig-johnson'||String(x?.title||'').trim().toLowerCase()==='sana tu cuerpo')return '/sana-tu-cuerpo-cover.jpg';
  return String(x?.cover_url||'')
};
