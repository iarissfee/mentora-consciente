function ebookCoverUrl(x){return x.cover_stored_name?`/api/ebooks/${x.id}/cover?v=${encodeURIComponent(x.updated_at||'')}`:x.cover_url}
function publicEbook(x){return{id:x.id,slug:x.slug,title:x.title,subtitle:x.subtitle,cover_url:ebookCoverUrl(x),price:x.price,currency:x.currency,ready:!!x.pdf_stored_name}}
function adminEbook(x){return{...x,display_cover_url:ebookCoverUrl(x)}}
function ebookForPurchase(id){const x=db.prepare('SELECT * FROM ebooks WHERE id=? AND active=1').get(Number(id));return x&&x.pdf_stored_name?x:null}
function hasEbookAccess(userId,ebookId){return !!db.prepare('SELECT 1 FROM ebook_access WHERE user_id=? AND ebook_id=?').get(userId,ebookId)}
function ebookCoverMime(name){const ext=path.extname(name||'').toLowerCase();if(ext==='.jpg'||ext==='.jpeg')return'image/jpeg';if(ext==='.png')return'image/png';if(ext==='.webp')return'image/webp';return'application/octet-stream'}
function validEbookCover(f){if(!f||!f.data||!f.data.length)return null;const type=String(f.contentType||'').toLowerCase();if(!['image/jpeg','image/png','image/webp'].includes(type))return null;const d=f.data,ok=type==='image/jpeg'?(d[0]===0xff&&d[1]===0xd8&&d[2]===0xff):type==='image/png'?(d.length>8&&d.slice(1,4).toString()==='PNG'):type==='image/webp'?(d.length>12&&d.slice(0,4).toString()==='RIFF'&&d.slice(8,12).toString()==='WEBP'):false;if(!ok)return null;return{ext:type==='image/jpeg'?'.jpg':type==='image/png'?'.png':'.webp',mime:type}}
