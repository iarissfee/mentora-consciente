function serveStatic(req,res,pathname){let rel=pathname==='/'?'index.html':pathname.replace(/^\/+/, '');let fp=path.resolve(PUBLIC_DIR,rel);if(!fp.startsWith(path.resolve(PUBLIC_DIR)+path.sep)&&fp!==path.resolve(PUBLIC_DIR,'index.html'))return false;if(fs.existsSync(fp)&&fs.statSync(fp).isFile()){securityHeaders(res);const st=fs.statSync(fp);res.writeHead(200,{'Content-Type':mime[path.extname(fp).toLowerCase()]||'application/octet-stream','Content-Length':st.size,'Cache-Control':isProd&&path.extname(fp)!=='.html'?'public, max-age=3600':'no-cache'});fs.createReadStream(fp).pipe(res);return true}return false}

const server=http.createServer(async(req,res)=>{try{const u=new URL(req.url,'http://local'),pathname=decodeURIComponent(u.pathname);if(pathname.startsWith('/api/')&&!rateOk(`api:${clientIp(req)}`,180,60000))return json(res,429,{error:'Demasiadas solicitudes.'});for(const r of routes){if(req.method!==r.method)continue;const m=pathname.match(r.re);if(!m)continue;const params={};r.keys.forEach((k,i)=>params[k]=decodeURIComponent(m[i+1]));return await r.handler(req,res,params,u)}if(req.method==='GET'||req.method==='HEAD'){if(serveStatic(req,res,pathname))return;const index=fs.readFileSync(path.join(PUBLIC_DIR,'index.html'));securityHeaders(res);res.writeHead(200,{'Content-Type':'text/html; charset=utf-8','Content-Length':index.length,'Cache-Control':'no-cache'});return res.end(index)}json(res,404,{error:'No encontrado.'})}catch(e){console.error(e);json(res,e.status||500,{error:e.status?e.message:'Ocurrió un error inesperado.'})}});
server.listen(PORT,'0.0.0.0',()=>console.log(`Campus listo en http://0.0.0.0:${PORT}`));
