route('POST','/api/admin/courses/:id/assets',async(req,res,p)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;const cid=Number(p.id);if(!db.prepare('SELECT id FROM courses WHERE id=?').get(cid))return json(res,404,{error:'Curso no encontrado.'});const ct=String(req.headers['content-type']||'');if(!ct.startsWith('multipart/form-data'))return json(res,400,{error:'Formato de carga inválido.'});let buf;try{buf=await readBody(req,6*1024*1024+200000)}catch{return json(res,400,{error:'El PDF supera el límite de 6 MB.'})}const {fields,files}=parseMultipart(buf,ct),f=files.pdf;if(!f)return json(res,400,{error:'Elegí un PDF.'});if(f.data.length>6*1024*1024)return json(res,400,{error:'El PDF supera el límite de 6 MB.'});if(f.contentType!=='application/pdf'||path.extname(f.filename).toLowerCase()!=='.pdf'||f.data.slice(0,5).toString()!=='%PDF-')return json(res,400,{error:'El archivo no parece ser un PDF válido.'});let lessonId=null;if(String(fields.lesson_id||'').trim()){lessonId=Number(fields.lesson_id);const lesson=db.prepare('SELECT id FROM lessons WHERE id=? AND course_id=?').get(lessonId,cid);if(!lesson)return json(res,400,{error:'La clase seleccionada no pertenece a este curso.'})}const stored=`${Date.now()}-${crypto.randomBytes(12).toString('hex')}.pdf`;fs.writeFileSync(path.join(UPLOAD_DIR,stored),f.data,{mode:0o600});const r=db.prepare('INSERT INTO assets(course_id,lesson_id,title,stored_name,original_name,mime,bytes,required_rank,downloadable,created_at) VALUES(?,?,?,?,?,?,?,?,?,?)').run(cid,lessonId,safeString(fields.title||f.filename,140),stored,path.basename(f.filename),'application/pdf',f.data.length,Math.max(0,Math.min(9,Number(fields.required_rank||0))),fields.downloadable==='false'?0:1,nowIso());json(res,200,{ok:true,id:Number(r.lastInsertRowid)})});
