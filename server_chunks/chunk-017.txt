
route('GET','/api/admin/community',async(req,res)=>{if(!auth(req,res,true))return;const rows=db.prepare("SELECT p.*,u.name author_name,u.email author_email,u.role author_role,(SELECT COUNT(*) FROM community_likes l WHERE l.post_id=p.id) likes_count,(SELECT COUNT(*) FROM community_comments c WHERE c.post_id=p.id AND c.status='published') comments_count FROM community_posts p JOIN users u ON u.id=p.user_id ORDER BY p.pinned DESC,p.id DESC LIMIT 200").all();json(res,200,{posts:rows,totals:{posts:rows.length,published:rows.filter(x=>x.status==='published').length,hidden:rows.filter(x=>x.status==='hidden').length}})});
route('POST','/api/admin/community/posts',async(req,res)=>{const u=auth(req,res,true);if(!u||!stateGuard(req,res))return;const b=await readJson(req),body=safeString(b.body,3000);if(body.length<2)return json(res,400,{error:'Escribí el contenido de la publicación.'});const r=db.prepare('INSERT INTO community_posts(user_id,body,image_url,space,status,pinned,created_at,updated_at) VALUES(?,?,?,?,?,?,?,?)').run(u.id,body,cleanCommunityImage(b.image_url),safeString(b.space||'Anuncios',60),b.status==='hidden'?'hidden':'published',b.pinned?1:0,nowIso(),nowIso());json(res,200,{ok:true,id:Number(r.lastInsertRowid)})});
route('PUT','/api/admin/community/posts/:id',async(req,res,p)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;const id=Number(p.id),post=db.prepare('SELECT * FROM community_posts WHERE id=?').get(id);if(!post)return json(res,404,{error:'Publicación no encontrada.'});const b=await readJson(req);db.prepare('UPDATE community_posts SET body=?,image_url=?,space=?,status=?,pinned=?,updated_at=? WHERE id=?').run(safeString(b.body??post.body,3000),b.image_url===undefined?post.image_url:cleanCommunityImage(b.image_url),safeString(b.space??post.space,60),b.status==='hidden'?'hidden':'published',b.pinned===undefined?post.pinned:(b.pinned?1:0),nowIso(),id);json(res,200,{ok:true})});
route('DELETE','/api/admin/community/posts/:id',async(req,res,p)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;db.prepare('DELETE FROM community_posts WHERE id=?').run(Number(p.id));json(res,200,{ok:true})});
