
function cleanCommunityImage(v){const s=safeString(v,1000);if(!s)return '';try{const u=new URL(s);return (u.protocol==='https:'||u.protocol==='http:')?s:''}catch{return ''}}
function communityPostRow(row){const comments=db.prepare("SELECT c.id,c.body,c.created_at,u.name author_name,u.role author_role FROM community_comments c JOIN users u ON u.id=c.user_id WHERE c.post_id=? AND c.status='published' ORDER BY c.id ASC LIMIT 50").all(row.id);return {...row,liked:!!row.liked,comments}}
route('PUT','/api/admin/courses/:id',async(req,res,p)=>{if(!auth(req,res,true)||!stateGuard(req,res))return;const id=Number(p.id),course=db.prepare('SELECT * FROM courses WHERE id=?').get(id);if(!course)return json(res,404,{error:'Programa no encontrado.'});const b=await readJson(req),title=safeString(b.title??course.title,120),slug=safeString(b.slug??course.slug,120).toLowerCase().replace(/[^a-z0-9-]+/g,'-').replace(/^-|-$/g,''),subtitle=safeString(b.subtitle??course.subtitle,180),description=safeString(b.description??course.description,1800),price=money(b.price??course.price),currency=safeString(b.currency??course.currency,3).toUpperCase(),sort=Number(b.sort_order??course.sort_order),linked=!!db.prepare('SELECT id FROM home_programs WHERE course_id=?').get(id);if(title.length<3||slug.length<3)return json(res,400,{error:'Título y URL son obligatorios.'});try{db.prepare('UPDATE courses SET slug=?,title=?,subtitle=?,description=?,cover_url=?,price=?,currency=?,required_rank=?,allow_direct_purchase=?,published=?,sort_order=?,updated_at=? WHERE id=?').run(slug,title,subtitle,description,safeString(b.cover_url??course.cover_url,600),price,currency,linked?0:Math.max(0,Math.min(9,Number(b.required_rank??course.required_rank))),linked?1:(b.allow_direct_purchase===undefined?course.allow_direct_purchase:(b.allow_direct_purchase?1:0)),b.published===undefined?course.published:(b.published?1:0),sort,nowIso(),id);if(linked)db.prepare('UPDATE home_programs SET title=?,description=?,price=?,currency=?,button_label=?,sort_order=?,updated_at=? WHERE course_id=?').run(title,subtitle||description,price,currency,'Comprar',sort,nowIso(),id);json(res,200,{ok:true})}catch{return json(res,409,{error:'La URL del programa ya existe o los datos no son válidos.'})}});
route('GET','/api/community/posts',async(req,res)=>{const u=auth(req,res);if(!u)return;const rows=db.prepare("SELECT p.*,u.name author_name,u.role author_role,(SELECT COUNT(*) FROM community_likes l WHERE l.post_id=p.id) likes_count,(SELECT COUNT(*) FROM community_comments c WHERE c.post_id=p.id AND c.status='published') comments_count,EXISTS(SELECT 1 FROM community_likes l2 WHERE l2.post_id=p.id AND l2.user_id=?) liked FROM community_posts p JOIN users u ON u.id=p.user_id WHERE p.status='published' ORDER BY p.pinned DESC,p.id DESC LIMIT 100").all(u.id).map(communityPostRow);json(res,200,{posts:rows})});
route('POST','/api/community/posts',async(req,res)=>{const u=auth(req,res);if(!u||!stateGuard(req,res))return;if(!rateOk(`community-post:${u.id}`,12,15*60000))return json(res,429,{error:'Publicaste varias veces seguidas. Esperá unos minutos.'});const b=await readJson(req,2200000),body=safeString(b.body,3000);if(body.length<2)return json(res,400,{error:'Escribí algo antes de publicar.'});const r=db.prepare("INSERT INTO community_posts(user_id,body,image_url,space,status,pinned,created_at,updated_at) VALUES(?,?,?,?, 'published',0,?,?)").run(u.id,body,cleanCommunityImage(b.image_url),safeString(b.space||'General',60),nowIso(),nowIso());json(res,200,{ok:true,id:Number(r.lastInsertRowid)})});
route('POST','/api/community/posts/:id/like',async(req,res,p)=>{const u=auth(req,res);if(!u||!stateGuard(req,res))return;if(!rateOk(`community-like:${u.id}`,120,15*60000))return json(res,429,{error:'Demasiadas reacciones seguidas.'});const id=Number(p.id);if(!db.prepare("SELECT id FROM community_posts WHERE id=? AND status='published'").get(id))return json(res,404,{error:'Publicación no encontrada.'});const has=db.prepare('SELECT 1 FROM community_likes WHERE post_id=? AND user_id=?').get(id,u.id);if(has)db.prepare('DELETE FROM community_likes WHERE post_id=? AND user_id=?').run(id,u.id);else db.prepare('INSERT INTO community_likes(post_id,user_id,created_at) VALUES(?,?,?)').run(id,u.id,nowIso());json(res,200,{ok:true,liked:!has})});
route('POST','/api/community/posts/:id/comments',async(req,res,p)=>{const u=auth(req,res);if(!u||!stateGuard(req,res))return;if(!rateOk(`community-comment:${u.id}`,40,15*60000))return json(res,429,{error:'Demasiados comentarios seguidos. Esperá unos minutos.'});const id=Number(p.id);if(!db.prepare("SELECT id FROM community_posts WHERE id=? AND status='published'").get(id))return json(res,404,{error:'Publicación no encontrada.'});const b=await readJson(req),body=safeString(b.body,1200);if(body.length<1)return json(res,400,{error:'El comentario está vacío.'});const r=db.prepare("INSERT INTO community_comments(post_id,user_id,body,status,created_at) VALUES(?,?,?,'published',?)").run(id,u.id,body,nowIso());json(res,200,{ok:true,id:Number(r.lastInsertRowid)})});
