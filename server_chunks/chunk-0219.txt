
// Portada editable de Home desde Administración → Vista de Home y edición.
const __homeSettingsEditablePoster=homeSettings;
homeSettings=function(){
  const out=__homeSettingsEditablePoster();
  const s=settingsObj();
  const stored=String(s.home_video_poster_stored_name||'');
  if(stored)out.video_poster=`/api/home/custom-poster?v=${encodeURIComponent(stored)}`;
  return out
};

function homePosterMime(name){
  const ext=path.extname(name||'').toLowerCase();
  if(ext==='.png')return'image/png';
  if(ext==='.webp')return'image/webp';
  return'image/jpeg'
}

route('GET','/api/home/custom-poster',async(req,res)=>{
  const stored=String(settingsObj().home_video_poster_stored_name||'');
  if(!stored)return json(res,404,{error:'No hay una portada personalizada.'});
  const safe=path.basename(stored),fp=path.join(UPLOAD_DIR,safe);
  if(!fs.existsSync(fp)&&remotePool){
    try{
      const rr=await remotePool.query('SELECT data FROM mc_asset_blobs WHERE stored_name=$1',[safe]);
      if(rr.rows[0]?.data)fs.writeFileSync(fp,rr.rows[0].data,{mode:0o600});
    }catch(e){console.error('Recuperación portada Home:',e)}
  }
  if(!fs.existsSync(fp))return json(res,404,{error:'Portada no disponible.'});
  const st=fs.statSync(fp);securityHeaders(res);
  res.writeHead(200,{'Content-Type':homePosterMime(safe),'Content-Length':st.size,'Cache-Control':'public, max-age=31536000, immutable','Content-Disposition':'inline'});
  fs.createReadStream(fp).pipe(res)
});

route('POST','/api/admin/home/poster',async(req,res)=>{
  if(!auth(req,res,true)||!stateGuard(req,res))return;
  const ct=String(req.headers['content-type']||'');
  if(!ct.startsWith('multipart/form-data'))return json(res,400,{error:'Formato de carga inválido.'});
  let buf;try{buf=await readBody(req,8*1024*1024+250000)}catch{return json(res,400,{error:'La imagen supera el límite de 8 MB.'})}
  const {files}=parseMultipart(buf,ct),f=files.poster;
  if(!f)return json(res,400,{error:'Elegí una foto para la portada.'});
  if(f.data.length>8*1024*1024)return json(res,400,{error:'La imagen supera el límite de 8 MB.'});
  const info=validEbookCover(f);
  if(!info)return json(res,400,{error:'Usá una imagen JPG, PNG o WebP válida.'});
  const current=String(settingsObj().home_video_poster_stored_name||'');
  const stored=`home-poster-${Date.now()}-${crypto.randomBytes(8).toString('hex')}${info.ext}`;
  fs.writeFileSync(path.join(UPLOAD_DIR,stored),f.data,{mode:0o600});
  const q=db.prepare('INSERT INTO settings(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value');
  q.run('home_video_poster_stored_name',stored);
  if(current&&current!==stored){const old=path.join(UPLOAD_DIR,path.basename(current));if(fs.existsSync(old))fs.unlinkSync(old)}
  json(res,200,{ok:true,video_poster:`/api/home/custom-poster?v=${encodeURIComponent(stored)}`})
});

route('DELETE','/api/admin/home/poster',async(req,res)=>{
  if(!auth(req,res,true)||!stateGuard(req,res))return;
  const current=String(settingsObj().home_video_poster_stored_name||'');
  if(current){const fp=path.join(UPLOAD_DIR,path.basename(current));if(fs.existsSync(fp))fs.unlinkSync(fp)}
  db.prepare("INSERT INTO settings(key,value) VALUES('home_video_poster_stored_name','') ON CONFLICT(key) DO UPDATE SET value='' ").run();
  json(res,200,{ok:true,video_poster:'/api/home/original-poster'})
});
