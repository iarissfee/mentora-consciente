route('POST','/api/paypal/capture-order/:orderId',async(req,res,p)=>{const u=auth(req,res);if(!u||!stateGuard(req,res))return;try{await readJson(req);const orderId=safeString(p.orderId,80),pending=db.prepare('SELECT * FROM pending_orders WHERE provider_order_id=? AND user_id=?').get(orderId,u.id);if(!pending)return json(res,404,{error:'Orden no encontrada.'});const {token,base}=await paypalToken();const rr=await fetch(`${base}/v2/checkout/orders/${encodeURIComponent(orderId)}/capture`,{method:'POST',headers:{Authorization:`Bearer ${token}`,'Content-Type':'application/json','PayPal-Request-Id':crypto.randomUUID()},body:'{}'});const data=await rr.json();if(!rr.ok||data.status!=='COMPLETED')return json(res,409,{error:'El pago no quedó completado.',details:data.status||'UNKNOWN'});const cap=data.purchase_units?.[0]?.payments?.captures?.[0],amt=Number(cap?.amount?.value),cur=cap?.amount?.currency_code;if(money(amt)!==money(pending.amount)||cur!==pending.currency)return json(res,409,{error:'La validación del importe no coincide.'});grantPurchase(u.id,pending.item_type,pending.item_id,'paypal',orderId,pending.amount,pending.currency);db.prepare('UPDATE pending_orders SET status=? WHERE id=?').run('COMPLETED',pending.id);let redirect='/campus.html#campus';if(pending.item_type==='course'){const c=db.prepare('SELECT slug FROM courses WHERE id=?').get(pending.item_id);if(c)redirect=`/campus.html#course/${encodeURIComponent(c.slug)}`}json(res,200,{ok:true,redirect})}catch(e){console.error(e);json(res,502,{error:'No se pudo confirmar el pago.'})}});
