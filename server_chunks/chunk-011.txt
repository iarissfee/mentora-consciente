WHERE id=?').run('COMPLETED',pending.id)}}}}json(res,200,{received:true})}catch(e){console.error(e);json(res,500,{error:'Webhook error.'})}});

function parseMultipart(buf,contentType){const m=contentType.match(/boundary=(?:"([^"]+)"|([^;]+))/i);if(!m)throw Object.assign(new Error('Multipart inválido'),{status:400});const boundary=Buffer.from(`--${m[1]||m[2]}`),fields={},files={};let start=0;while(true){let a=buf.indexOf(boundary,start);if(a<0)break;a+=boundary.length;if(buf.slice(a,a+2).toString()==='--')break;if(buf.slice(a,a+2).toString()==='\r\n')a+=2;let b=buf.indexOf(boundary,a);if(b<0)break;let part=buf.slice(a,b-2),sep=part.indexOf(Buffer.from('\r\n\r\n'));if(sep<0){start=b;continue}const head=part.slice(0,sep).toString('utf8'),data=part.slice(sep+4);const nm=head.match(/name="([^"]+)"/i);if(!nm){start=b;continue}const fn=head.match(/filename="([^"]*)"/i),ct=(head.match(/content-type:\s*([^\r\n]+)/i)||[])[1]||'application/octet-stream';if(fn)files[nm[1]]={filename:path.basename(fn[1]),contentType:ct,data};else fields[nm[1]]=data.toString('utf8');start=b}return{fields,files}}
